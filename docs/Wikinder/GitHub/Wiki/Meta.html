Overview

Reads and writes META.md, syncing article rows with git history.

Existing rows are kept (author, creation date, aliases are curated by hand), but modified_date is refreshed from git and rows for missing or deleted pages are rebuilt from history. Language is detected once, when a row is created, and never refreshed — the detection is heuristic and flipping it after the fact would silently rewrite a value a human may already have reviewed.

Aliases that would shadow an existing page are dropped; see prune_colliding_aliases!.

With ignore_meta set, every row is rebuilt from history and the previously committed META.md is disregarded.

Cells are written with backslash escapes — '|' for a literal pipe and '\' for a literal backslash — so any author or title can be represented losslessly. Newlines inside a value are flattened to spaces. Hand-written rows without escapes are read leniently, and those tolerances are intentional rather than optional.

def self.build_row(wiki, page, bounds = nil, renames = nil)
  slug = File.basename(page, '.md')
  bounds ||= wiki.commit_bounds_for(page)
  renames ||= wiki.rename_history_for(page)
  last = bounds[:latest]
  first = bounds[:earliest]

  MetaRow.new(
    slug: slug,
    title: Slugs.title_for(slug),
    author: first ? first.author[:name] : 'unknown',
    creation_date: first ? first.time.utc.iso8601 : 'unknown',
    modified_date: last ? last.time.utc.iso8601 : 'unknown',
    language: Language.call(wiki.text_for(page), supported: wiki.supported_languages),
    aliases: renames.map { |name| Slugs.title_for(File.basename(name, '.md')) }
  )
end

def self.escape_cell(text)

def self.inherited_row(wiki, previously, page, slug)
  return previously[slug] if previously.key?(slug)

  wiki.rename_history_for(page).each do |predecessor|
    row = previously[File.basename(predecessor, '.md')]
    return row if row
  end
  nil
end

def self.load(meta_path, ignore_meta: false)
  return {} if ignore_meta
  return {} unless meta_path.exist?

  result = {} #: Hash[String, MetaRow]
  meta_path.each_line.with_index(1) do |line, index|
    # Strip invalid bytes up-front so the parser never raises
    # Encoding::CompatibilityError on a hand-corrupted line.
    line = line.scrub("\u{FFFD}")
    row = parse_row(line.chomp, index)
    next unless row

    raise Error, "META.md: duplicate row for [[#{row.slug}]] on line #{index}" if result.key?(row.slug)

    result[row.slug] = row
  end
  result
end

def self.parse_row(line, line_number = nil)
  location = line_number ? " (line #{line_number})" : ''
  cells = split_cells(line)
  return nil if cells.empty?
  return nil if cells.all? { |cell| cell.match?(/\A-+\z/) }
  # A single empty cell is a stray blank line, not a header row.
  return nil if cells.length == 1 && cells[0].empty?

  # The header row is a normal part of every META.md; skip it quietly.
  return nil if cells[0] == 'Title'

  raise Error, "META.md: row has too few cells#{location}" if cells.length < META_HEADERS.length

  if cells.length > META_HEADERS.length
    raise Error, "META.md: row has #{cells.length - META_HEADERS.length} extra cell(s)#{location}"
  end

  title_cell = cells[0]
  unless title_cell.start_with?('[[') && title_cell.end_with?(']]')
    raise Error, "META.md: row without a [[Title]] cell#{location}"
  end

  title = title_cell[2...-2].to_s.strip
  raise Error, "META.md: row with an empty [[]] cell#{location}" if title.empty?

  MetaRow.new(
    slug: Slugs.for_title(title),
    title: title,
    author: cells[1],
    creation_date: cells[2],
    modified_date: cells[3],
    language: cells[4],
    aliases: cells[5].split(ALIAS_SEPARATOR).map(&:strip).reject(&:empty?)
  )
end

.prune_colliding_aliases!(rows, pages, ungenerated_pages) ⇒ Object

Drops aliases that slugify to a name already used by an existing page (generated or ungenerated), or to the row's own slug.

An alias that resolves to the same slug as an existing page would produce a redirect file that shadows it — the existing page always wins. This also removes the stale alias from META.md itself, so a renamed-away name cannot be re-added by an automated pass, and a hand-curated alias whose name has since been claimed by another page is removed instead of shadowing it.

Additionally, an alias claimed by more than one page's row is dropped from every claimant and reported: a redirect file can only have one healthy target, and silently letting the last page win would hide the ambiguity regardless of write order.

Comparison is case-insensitive because the output target may live on a case-insensitive filesystem. : (Hash[String, MetaRow], Array, Set) -> void

def self.prune_colliding_aliases!(rows, pages, ungenerated_pages)
  occupied = Set.new(ungenerated_pages.map(&:downcase))
  pages.each { |page| occupied << File.basename(page, '.md').downcase }

  rows.each do |slug, row|
    kept = row.aliases.reject do |title|
      alias_slug = Slugs.for_title(title).downcase
      unless Slugs.safe?(alias_slug)
        warn "META.md: ignoring unsafe alias #{title.inspect}; it would not produce a valid page"
        next true
      end
      alias_slug == row.slug.downcase || occupied.include?(alias_slug)
    end
    rows[slug] = row.with(aliases: kept)
  end

  claims = Hash.new { |hash, key| hash[key] = [] } #: Hash[String, Array[[String, String]]]
  rows.each do |slug, row|
    row.aliases.each do |title|
      claims[Slugs.for_title(title).downcase] << [slug, title]
    end
  end
  claims.each_value do |pairs|
    claimants = pairs.map(&:first).uniq
    next if claimants.length <= 1

    # Use a stable owner list for the warning so the same set of
    # pages always produces the same diagnosis.
    owners = claimants.map { |slug| "[[#{slug}]]" }.sort.join(', ')
    warn "META.md: alias #{pairs.first[1].inspect} is claimed by #{owners}; ignoring it"
    pairs.group_by(&:first).each do |slug, owned|
      row = rows.fetch(slug)
      rows[slug] = row.with(aliases: row.aliases - owned.map(&:last))
    end
  end
end

def self.refresh_row(wiki, page, previous)
  slug = File.basename(page, '.md')
  bounds = wiki.commit_bounds_for(page)
  last = bounds[:latest]
  renames = wiki.rename_history_for(page)
  aliases = renames.map { |name| Slugs.title_for(File.basename(name, '.md')) }

  if previous
    return previous.with(
      slug: slug,
      title: Slugs.title_for(slug),
      modified_date: last ? last.time.utc.iso8601 : previous.modified_date,
      aliases: merge_aliases(previous.aliases, aliases)
    )
  end

  build_row(wiki, page, bounds, renames)
end

def self.split_cells(line)
  body = line.chomp
  return [] unless body.start_with?('|') && body.end_with?('|')

  inner = body[1..-2] || ''
  cells = [] #: Array[String]
  current = +''
  i = 0
  while i < inner.length
    ch = inner[i]
    break if ch.nil? # cannot happen, satisfies the type checker

    if ch == '\\'
      nxt = inner[i + 1]
      if nxt && ['\\', '|'].include?(nxt)
        current << nxt
        i += 2
        next
      end
      current << ch
      i += 1
      next
    end
    if ch == '|'
      cells << current
      current = +''
      i += 1
      next
    end
    current << ch
    i += 1
  end
  cells << current
  cells.map(&:strip)
end

def self.sync(wiki, pages, meta_path, ignore_meta: false, prev_head: nil)
  # diff_paths returns nil for "full rebuild" (ignore_meta or no valid
  # prev_head); it returns the exact list of paths touched since
  # prev_head otherwise.
  previously = load(meta_path, ignore_meta: ignore_meta)
  rows = {} #: Hash[String, MetaRow]

  changed = prev_head.nil? || ignore_meta ? nil : wiki.diff_paths(prev_head)

  if changed && changed.empty?
    # No page changed since prev_head, but META.md itself may
    # have drifted (reverted or hand-edited). Keep the rows we
    # already have, but rebuild rows for pages they no longer
    # cover — via refresh_row, so rename inheritance and
    # curated fields are honoured exactly as in a full rebuild,
    # not dropped to git-derived defaults.
    still_current = Set.new(pages.map { |page| File.basename(page, '.md') })
    previously.each { |slug, row| rows[slug] = row }
    pages.each do |page|
      slug = File.basename(page, '.md')
      rows[slug] ||= refresh_row(wiki, page, inherited_row(wiki, previously, page, slug))
    end
    rows.delete_if { |slug, _| !still_current.include?(slug) }
    prune_colliding_aliases!(rows, pages, wiki.ungenerated_pages)
    return rows
  end

  pages.each do |page|
    slug = File.basename(page, '.md')
    inherited = inherited_row(wiki, previously, page, slug)
    if !changed.nil? && inherited && changed.none? { |path| path.b == page.b }
      rows[slug] = inherited.with(slug: slug, title: Slugs.title_for(slug))
      next
    end

    rows[slug] = refresh_row(wiki, page, inherited)
  end

  prune_colliding_aliases!(rows, pages, wiki.ungenerated_pages)
  rows
end

def self.write(meta_path, rows)
  sorted = rows.values.sort do |a, b|
    (b.creation_date <=> a.creation_date).nonzero? ||
      (a.slug <=> b.slug)
  end
  lines = [*META_TABLE_HEADER]
  sorted.each do |row|
    title = row.title
    cells = [
      "[[#{escape_cell(title)}]]",
      escape_cell(row.author),
      escape_cell(row.creation_date),
      escape_cell(row.modified_date),
      escape_cell(row.language),
      escape_cell(row.aliases.join(ALIAS_SEPARATOR))
    ]
    inner = cells.map { |cell| cell.empty? ? ' ' : " #{cell} " }.join('|')
    lines << "|#{inner}|"
  end
  content = lines.join("\n") << "\n"
  return if meta_path.exist? && meta_path.read == content

  write_atomically(meta_path, content)
end

def self.write_atomically(meta_path, content)
  temp = "#{meta_path}.tmp.#{Process.pid}"
  begin
    File.write(temp, content)
    File.rename(temp, meta_path)
  ensure
    FileUtils.rm_f(temp)
  end
end
